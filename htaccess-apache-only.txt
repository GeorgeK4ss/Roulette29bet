Options -MultiViews
RewriteEngine On

# Canonical host: force HTTPS and strip www
RewriteCond %{HTTPS} !=on
RewriteCond %{HTTP:X-Forwarded-Proto} !=https
RewriteRule ^ https://www.roulettewheelbonus.com%{REQUEST_URI} [R=301,L]
# The apex hosts a different site, so canonical host is www (not the usual
# www-stripping redirect).
RewriteCond %{HTTP_HOST} !^www\. [NC]
RewriteRule ^ https://www.roulettewheelbonus.com%{REQUEST_URI} [R=301,L]

# --- Markdown for AI agents ---------------------------------------
# An agent sending "Accept: text/markdown" gets index.md.
# Browsers send "Accept: text/html,..." and never match this rule.
AddType text/markdown .md

RewriteCond %{HTTP_ACCEPT} text/markdown [NC]
RewriteCond %{REQUEST_URI} !\.md$
RewriteCond %{DOCUMENT_ROOT}/$1index.md -f
RewriteRule ^(.*)$ /$1index.md [L]
# ------------------------------------------------------------------

# --- Compression ---
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/plain text/css text/xml
  AddOutputFilterByType DEFLATE application/javascript application/json application/xml
  AddOutputFilterByType DEFLATE image/svg+xml text/markdown
</IfModule>

# --- Caching ---
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType text/css "access plus 1 year"
  ExpiresByType application/javascript "access plus 1 year"
  ExpiresByType image/webp "access plus 6 months"
  ExpiresByType image/png "access plus 6 months"
  ExpiresByType image/jpeg "access plus 6 months"
  ExpiresByType image/svg+xml "access plus 6 months"
  # HTML and its Markdown twin must always revalidate
  ExpiresByType text/html "access plus 0 seconds"
  ExpiresByType text/markdown "access plus 0 seconds"
</IfModule>

<IfModule mod_headers.c>
  # CSS/JS are versioned with ?v= so they can cache hard
  <FilesMatch "\.(css|js|woff2)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
  <FilesMatch "\.(html|md)$">
    Header set Cache-Control "no-cache, must-revalidate"
  </FilesMatch>

  <FilesMatch "\.md$">
    Header set Content-Type "text/markdown; charset=utf-8"
  </FilesMatch>

  # Same URL returns HTML or Markdown depending on Accept — without this
  # a cache can serve the wrong one.
  <FilesMatch "\.(html|md)?$">
    Header append Vary Accept
  </FilesMatch>

  # --- Security headers ---
  Header always set X-Content-Type-Options "nosniff"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set Permissions-Policy "geolocation=(), microphone=(), camera=()"
  Header always set Strict-Transport-Security "max-age=63072000; includeSubDomains; preload"
  Header always set Cross-Origin-Opener-Policy "same-origin"

  # CSP scoped to what this page actually loads:
  #   Google Fonts (stylesheet + font files), Meta Pixel (script + tracking
  #   pixel + beacon). Nothing else is permitted.
  Header always set Content-Security-Policy "default-src 'self'; script-src 'self' 'unsafe-inline' https://connect.facebook.net; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com; font-src 'self' data: https://fonts.gstatic.com; img-src 'self' data: https://www.facebook.com; connect-src 'self' https://www.facebook.com; frame-src https://www.facebook.com; object-src 'none'; base-uri 'self'; form-action 'self' https://www.facebook.com; frame-ancestors 'self'; upgrade-insecure-requests"
</IfModule>

# Real 404s
ErrorDocument 404 /index.html
